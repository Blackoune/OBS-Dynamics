# Versions ÉPINGLÉES (2026-09-09). Sans `==`, deux installations faites à deux
# dates différentes ne donnent pas le même logiciel : une version majeure
# publiée entre-temps casse le build ou change un comportement, et le rapport
# de bug devient impossible à reproduire. Ce sont les versions sous lesquelles
# les 380 tests passent.
#
# Pour monter une version : la changer ici, relancer `pytest`, puis `build.py`.

customtkinter==6.0.0        # interface graphique
opencv-python==5.0.0.93     # détection visuelle (matchTemplate)
numpy==2.5.1                # manipulé directement par detection.py et cover_service
Pillow==12.3.0              # capture d'écran (ImageGrab) + traitement des jaquettes
psutil==7.2.2               # détection des processus de jeu
simpleobsws==1.4.3          # client OBS WebSocket v5
requests==2.34.2            # téléchargement des jaquettes (cover_service)
pynput==1.8.2               # hotkeys globales (dégradation propre si absent)

# Widget Musique : lecture des métadonnées « en cours de lecture » publiées par
# Windows (SMTC). Le paquet `winsdk`, cité par presque tous les exemples en
# ligne, n'a PAS de roue pour Python 3.14 : ce sont les projections `winrt-*`
# qu'il faut. `winrt-Windows.Foundation.Collections` n'est importé par aucun
# de nos fichiers, mais la projection le charge à la volée dès le premier
# `get_sessions()` — sans lui, ModuleNotFoundError au premier morceau lu.
# `music_smtc` dégrade proprement si ces paquets manquent : l'onglet affiche
# l'erreur et la commande d'installation, le reste de l'application fonctionne.
winrt-Windows.Media.Control==3.2.1
winrt-Windows.Storage.Streams==3.2.1
winrt-Windows.Foundation==3.2.1
winrt-Windows.Foundation.Collections==3.2.1
winrt-runtime==3.2.1

# Forme d'onde du widget musique : niveau audio PAR APPLICATION, lu sur le
# compteur de la session (`IAudioMeterInformation`) — le meme que le melangeur
# de volume de Windows. C'est ce qui permet a l'overlay Spotify de ne reagir
# qu'au son de Spotify. Windows uniquement ; `music_audio.available()` est faux
# ailleurs et l'overlay fonctionne sans forme d'onde.
pycaw==20251023
comtypes==1.4.16
tkinterdnd2==0.6.3          # glisser-déposer de médias sur la zone de dépôt
